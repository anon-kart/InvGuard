# DEFAULT FLOW-2 INPUT. Fail-closed if empty.
(meta.revert_taken == false) -> (select(post.balances, meta.sender) <= (select(pre.balances, meta.sender) + meta.value))
(meta.revert_taken == false) -> (select(post.balances, meta.sender) == (select(pre.balances, meta.sender) + meta.value))
(meta.revert_taken == false) -> (select(post.balances, meta.sender) >= select(pre.balances, meta.sender))
(meta.revert_taken == false) -> (post.salt == pre.salt)
(meta.revert_taken == false) -> (select(post.balances, meta.sender) == select(pre.balances, meta.sender) + (select(post.balances, meta.sender) - select(pre.balances, meta.sender)))
